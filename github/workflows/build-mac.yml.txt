name: Build Mac App

on:
  push:
    branches: [ main ]

jobs:
  build:
    runs-on: macos-latest

    steps:
    - name: Kodları İndir
      uses: actions/checkout@v3

    - name: Python Kur
      uses: actions/setup-python@v4
      with:
        python-version: '3.10'

    - name: FFmpeg Kur (Mac için)
      run: brew install ffmpeg

    - name: Bağımlılıkları Yükle
      run: |
        cd backend
        python -m venv venv
        source venv/bin/activate
        pip install -r requirements.txt
        pip install pyinstaller

    - name: PyInstaller ile Mac Uygulaması Üret
      run: |
        cd backend
        source venv/bin/activate
        pyinstaller --onedir --add-data "dist:dist" main.py

    - name: Dosyayı Arşivle (.zip yap)
      run: |
        cd backend/dist
        zip -r ../../kalia-mac.zip main.app
        
    - name: Üretilen Mac Dosyasını İndirilebilir Yap
      uses: actions/upload-artifact@v3
      with:
        name: Mac-App
        path: kalia-mac.zip
